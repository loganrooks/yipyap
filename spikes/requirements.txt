# Spike-only dependencies. NOT project dependencies — these never enter
# pyproject.toml. The whole spikes/ directory is throwaway per docs/spike-plan.md.
#
# Install into a dedicated venv:
#   python3.11 -m venv .venv-spikes
#   source .venv-spikes/bin/activate
#   pip install -r spikes/requirements.txt
#
# Loose pins on purpose — this is throwaway and we want the latest compatible
# release at install time, not a frozen lockfile.

demucs>=4.0.1
librosa>=0.10.1
soundfile>=0.12.1
numpy>=1.24
scipy>=1.10
silero-vad>=5.0
# Spike A separation backends. python-audio-separator (PyPI name:
# audio-separator) is the new cross-family method (MDX-Net via UVR). It pulls
# numpy>=2 transitively, so we drop the upper pin on numpy; librosa 0.11 and
# demucs 4.0.1 both handle numpy 2.x.
audio-separator[cpu]>=0.18
# Open-Unmix is already installed transitively as a Demucs dep, but listing it
# explicitly so the umx backend is a first-class spike method.
openunmix>=1.2.1
# ASR for the transcription-based Spike B fallback (spike-plan documents this
# as the path to take if onset/cadence synthesis doesn't sound enough like
# animalese). mlx-whisper runs natively on Apple Silicon's Metal GPU and is
# the primary path; faster-whisper is a portable CPU fallback for Linux /
# Intel Macs.
mlx-whisper>=0.4 ; sys_platform == "darwin" and platform_machine == "arm64"
faster-whisper>=1.0
# Speaker diarization for the ASR-driven Spike B path. pyannote uses speaker
# embeddings (timbre, not pitch), so a single excited commentator whose
# pitch jumps an octave on a podium finish stays in one cluster. Requires
# HF_TOKEN env var at runtime and acceptance of model conditions at
# https://hf.co/pyannote/speaker-diarization-3.1.
pyannote.audio>=4.0
# torch and torchaudio come transitively via demucs; pinning them here would
# fight demucs's own constraint solver. Leave them implicit.
#
# DeepFilterNet is intentionally NOT a hard requirement. It is an OPTIONAL
# Phase-0 fallback (speech enhancement, not separation). To try it locally:
#   pip install deepfilternet
# See "Trying DeepFilterNet (optional)" in spikes/README.md.
